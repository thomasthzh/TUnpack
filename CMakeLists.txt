cmake_minimum_required(VERSION 3.24)
project(TUnpack VERSION 1.0.0 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(BIT7Z_AUTO_FORMAT ON CACHE BOOL "" FORCE)
set(BIT7Z_DETECT_FROM_EXTENSION OFF CACHE BOOL "" FORCE)
set(BIT7Z_USE_NATIVE_STRING ON CACHE BOOL "" FORCE)
set(BIT7Z_DISABLE_USE_STD_FILESYSTEM ON CACHE BOOL "" FORCE)
set(BIT7Z_AUTO_PREFIX_LONG_PATHS ON CACHE BOOL "" FORCE)
set(BIT7Z_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory(tools/bit7z EXCLUDE_FROM_ALL)
# Keep archive paths inside the build tree so Ninja passes ASCII relative names to MinGW ar.
set_target_properties(bit7z64 PROPERTIES ARCHIVE_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(MINGW)
  target_compile_options(bit7z64 PRIVATE -ffunction-sections -fdata-sections -Os)
  if(CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 16)
    # GCC 16's speculative COM devirtualization misidentifies 7z stream subclasses.
    target_compile_options(bit7z64 PRIVATE -fno-devirtualize -fno-devirtualize-speculatively)
  endif()
endif()
add_library(unpack_core src/input.cpp src/core.cpp src/passwords.cpp src/password_probe.cpp src/storage.cpp)
target_include_directories(unpack_core PUBLIC src)
target_include_directories(unpack_core SYSTEM PUBLIC "${CMAKE_BINARY_DIR}/_deps/ghc_filesystem-src/include")
target_compile_definitions(unpack_core PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_link_libraries(unpack_core PUBLIC bit7z crypt32 shell32 ole32)
configure_file(src/app.manifest "${CMAKE_BINARY_DIR}/app.manifest" COPYONLY)
configure_file(brand/t-unpack.ico "${CMAKE_BINARY_DIR}/t-unpack.ico" COPYONLY)
if(MINGW)
  # windres' narrow popen fails on a Chinese absolute input path; keep its arguments ASCII/relative.
  configure_file(src/app.rc "${CMAKE_BINARY_DIR}/app.rc" COPYONLY)
  add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/app.rc.obj"
    COMMAND "${CMAKE_RC_COMPILER}" --use-temp-file -O coff app.rc app.rc.obj
    DEPENDS src/app.rc brand/t-unpack.ico src/app.manifest
    WORKING_DIRECTORY "${CMAKE_BINARY_DIR}" VERBATIM)
  set(APP_RESOURCE "${CMAKE_BINARY_DIR}/app.rc.obj")
else()
  add_library(app_resources OBJECT src/app.rc)
  set(APP_RESOURCE $<TARGET_OBJECTS:app_resources>)
endif()
add_library(unpack_ui_model src/ui_model.cpp src/ui_layout.cpp src/queue_runner.cpp src/workspace.cpp src/shell_open.cpp)
target_link_libraries(unpack_ui_model PUBLIC unpack_core shlwapi gdi32 advapi32)
add_library(unpack_drop src/file_drop.cpp)
target_link_libraries(unpack_drop PUBLIC unpack_core uuid)
add_executable(TUnpack WIN32 src/app.cpp ${APP_RESOURCE})
if(MINGW)
  # MinGW ld writes non-ASCII output names using the wrong code page. Copy through CMake's Unicode API.
  add_custom_command(TARGET TUnpack POST_BUILD
    COMMAND "${CMAKE_COMMAND}" -E copy "$<TARGET_FILE:TUnpack>" "${CMAKE_BINARY_DIR}/bin/T解压.exe"
    BYPRODUCTS "${CMAKE_BINARY_DIR}/bin/T解压.exe" VERBATIM)
else()
  set_target_properties(TUnpack PROPERTIES OUTPUT_NAME "T解压")
endif()
target_link_libraries(TUnpack PRIVATE unpack_ui_model unpack_drop comctl32 comdlg32 uxtheme gdi32 shlwapi)
add_executable(unpack_cli src/cli.cpp)
target_link_libraries(unpack_cli PRIVATE unpack_core)
add_executable(unpack_checks tests/core_checks.cpp)
target_link_libraries(unpack_checks PRIVATE unpack_core)
add_executable(ui_checks tests/ui_checks.cpp)
target_link_libraries(ui_checks PRIVATE unpack_ui_model)
add_executable(drop_checks tests/drop_checks.cpp)
target_link_libraries(drop_checks PRIVATE unpack_drop unpack_ui_model)
add_executable(flow_checks tests/flow_checks.cpp)
target_link_libraries(flow_checks PRIVATE unpack_ui_model)
add_executable(layout_checks tests/layout_checks.cpp)
target_link_libraries(layout_checks PRIVATE unpack_ui_model)
add_executable(workspace_checks tests/workspace_checks.cpp)
target_link_libraries(workspace_checks PRIVATE unpack_ui_model)
add_executable(open_checks tests/open_checks.cpp)
target_link_libraries(open_checks PRIVATE unpack_ui_model)
add_executable(resource_checks tests/resource_checks.cpp)
target_compile_definitions(resource_checks PRIVATE UNICODE _UNICODE NOMINMAX)
target_link_libraries(resource_checks PRIVATE user32 gdi32 version)
if(MINGW)
  foreach(_t TUnpack unpack_cli unpack_checks ui_checks drop_checks flow_checks workspace_checks open_checks resource_checks)
    target_link_options(${_t} PRIVATE -municode -static -static-libgcc -static-libstdc++ -s -Wl,--gc-sections)
  endforeach()
  target_link_options(layout_checks PRIVATE -static -static-libgcc -static-libstdc++ -s -Wl,--gc-sections)
  target_compile_options(unpack_core PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-O3>)
  target_compile_options(unpack_ui_model PRIVATE -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-Os>)
  target_compile_options(unpack_drop PRIVATE -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-Os>)
  target_compile_options(TUnpack PRIVATE -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-Os>)
  target_compile_options(unpack_cli PRIVATE -ffunction-sections -fdata-sections $<$<CONFIG:Release>:-Os>)
endif()
set_target_properties(TUnpack unpack_cli unpack_checks ui_checks drop_checks flow_checks layout_checks workspace_checks open_checks resource_checks PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(UNPACK_ENGINE "${CMAKE_SOURCE_DIR}/tools/7zip-zs-26.05/7z.dll" CACHE FILEPATH "7z shared library")
get_filename_component(UNPACK_ENGINE_DIR "${UNPACK_ENGINE}" DIRECTORY)
configure_file("${UNPACK_ENGINE}" "${CMAKE_BINARY_DIR}/bin/7z.dll" COPYONLY)
configure_file("${UNPACK_ENGINE_DIR}/License.txt" "${CMAKE_BINARY_DIR}/bin/7-Zip-License.txt" COPYONLY)
configure_file("${CMAKE_SOURCE_DIR}/tools/bit7z/LICENSE" "${CMAKE_BINARY_DIR}/bin/bit7z-LICENSE.txt" COPYONLY)
enable_testing()
add_test(NAME brand_resources COMMAND resource_checks "${CMAKE_BINARY_DIR}/bin/T解压.exe")
add_test(NAME ui_model COMMAND ui_checks "${CMAKE_SOURCE_DIR}/test-data")
add_test(NAME native_drop COMMAND drop_checks "${CMAKE_SOURCE_DIR}/test-data")
add_test(NAME queue_flows COMMAND flow_checks "${CMAKE_SOURCE_DIR}/test-data" "${CMAKE_SOURCE_DIR}/tests/fixtures")
add_test(NAME layout_dpi COMMAND layout_checks)
add_test(NAME workspace COMMAND workspace_checks "${CMAKE_SOURCE_DIR}/test-data")
add_test(NAME shell_open COMMAND open_checks "${CMAKE_SOURCE_DIR}/test-data")
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" -X utf8 "${CMAKE_SOURCE_DIR}/tests/integration.py")
  add_test(NAME zip_native_volumes COMMAND "${Python3_EXECUTABLE}" -X utf8 "${CMAKE_SOURCE_DIR}/tests/zip_volumes.py")
endif()
